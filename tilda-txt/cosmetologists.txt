(function(){
  var root=document.getElementById('calyra-lp'); if(!root) return;
  var hasIO='IntersectionObserver' in window;

  /* блок «что даёт»: пункты проявляются по очереди */
  var why=document.getElementById('clWhy');
  if(hasIO){ var wo=new IntersectionObserver(function(es){ if(es[0].isIntersecting){ why.classList.add('in'); wo.disconnect(); } },{threshold:.3}); wo.observe(why); } else { why.classList.add('in'); }

  /* слои кожи: сцена закреплена, шаги меняются по прокрутке */
  var skin=document.getElementById('clSkin'), scene=document.getElementById('clScene');
  var steps=root.querySelectorAll('.cl-step'), prog=root.querySelectorAll('#clProg button'), act=-1;
  function setStep(i){ if(i===act) return; act=i;
    skin.setAttribute('data-active',i+1);
    steps.forEach(function(s,k){s.classList.toggle('on',k===i)});
    prog.forEach(function(b,k){b.classList.toggle('done',k<=i)}); }
  function sceneScroll(){ var r=scene.getBoundingClientRect(), tot=r.height-window.innerHeight;
    var p=Math.min(Math.max(-r.top/tot,0),.999); setStep(Math.floor(p*3)); }
  prog.forEach(function(b,k){ b.addEventListener('click',function(){
    var top=scene.getBoundingClientRect().top+window.pageYOffset, tot=scene.offsetHeight-window.innerHeight;
    window.scrollTo({top:top+tot*(k/3+.12),behavior:'smooth'}); }); });
  window.addEventListener('scroll',sceneScroll,{passive:true});
  window.addEventListener('resize',sceneScroll);
  sceneScroll();

  /* схема пептида и счётчики */
  var radial=document.getElementById('clRadial');
  var nums=root.querySelectorAll('#clNums b[data-to]');
  function count(el){
    var to=+el.getAttribute('data-to'), suf=el.getAttribute('data-suffix')||'', t0=null;
    if(window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    function f(t){ if(!t0)t0=t; var p=Math.min((t-t0)/1400,1); el.textContent=Math.round(to*(1-Math.pow(1-p,3)))+(p===1?suf:''); if(p<1)requestAnimationFrame(f); }
    requestAnimationFrame(f);
  }
  if(hasIO){
    var oo=new IntersectionObserver(function(es){
      es.forEach(function(e){
        if(!e.isIntersecting) return;
        if(e.target===radial) radial.classList.add('on'); else nums.forEach(count);
        oo.unobserve(e.target);
      });
    },{threshold:.35});
    oo.observe(radial); oo.observe(document.getElementById('clNums'));
  } else { radial.classList.add('on'); }

  /* шкала курса + липкая кнопка на мобильном */
  var track=document.getElementById('clTrack'), fill=document.getElementById('clTrackFill');
  var sticky=document.getElementById('clSticky'), hero=document.getElementById('cl-top'), finalSec=root.querySelector('.cl-final');
  function onScroll(){
    var r=track.getBoundingClientRect(), vh=window.innerHeight;
    var p=Math.max(0,Math.min(1,(vh*0.8-r.top)/(r.height+vh*0.3)));
    fill.style.transform=(window.innerWidth<=900)?'scaleY('+p+')':'scaleX('+p+')';
    var past=hero.getBoundingClientRect().bottom<0;
    var atEnd=finalSec.getBoundingClientRect().top<vh;
    sticky.classList.toggle('show',past&&!atEnd);
  }
  window.addEventListener('scroll',onScroll,{passive:true});
  window.addEventListener('resize',onScroll);

  function startLad(){ lad.classList.add('in'); if(window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    timer=setInterval(function(){ setLvl((cur+1)%3); },3200); }
  if(hasIO){ var lo=new IntersectionObserver(function(es){ if(es[0].isIntersecting){ startLad(); lo.disconnect(); } },{threshold:.4}); lo.observe(lad); } else { lad.classList.add('in'); }

  });
